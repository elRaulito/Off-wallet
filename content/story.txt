# OffWallet content file
# =======================
# Everything the game says, shows and who says it. Edit this, then rebuild:
#
#     bash tools/build.sh game
#
# tools/gen/gen_story.py compiles this into the ROM and CHECKS IT for you:
# lines that are too wide, missing sections, unknown ids and art that is not
# 16x16 all stop the build with a clear message. A Game Boy screen is 20 tiles
# wide and 18 tall, so:
#
#   * screen text starting at column x can be at most (20 - x) characters
#   * dialogue lines can be at most 18 characters (the box eats two columns)
#
# Syntax
# ------
#   # comment
#   [section id]        starts a section; keys are "key = value"
#
# Sections
# --------
#   [config]            a few positions and the unlocked starter
#   [palette id]        colors = #RRGGBB x4  (background, light, mid, dark)
#   [sprite id]         16 lines of 16 characters: . = colour 0, - = 1,
#                       + = 2, # = 3. Becomes a 16x16 (2x2 tile) graphic.
#   [character id]      name / sprite / palette / portrait = x,y
#   [screen id]         a list of draw commands, one per line:
#                         text   x,y <string>     normal colours
#                         accent x,y <string>     highlight palette
#                         warn   x,y <string>     red palette
#                         sprite x,y <sprite id>
#                         blank  x,y,width        erase a run of tiles
#                       key "clear = no" draws on top of what is already there
#   [dialog id]         one or more boxes. "speaker = <character id>" applies
#                       to the boxes that follow; "--" starts the next box;
#                       up to 3 lines per box.
#
# Built-in palettes: text, warn, accent. All artwork is declared below.
# Section ids marked REQUIRED below are referenced by the game code: you can
# change their contents freely, but do not rename or delete them.

# Sprites drawn as PNGs bring their own colours (see [sprite] below), so the
# only palettes needed here are ones you want to name and reuse, for instance
# for text. Three are built in: text, warn and accent.


# ------------------------------------------------------------------ sprites
# Artwork lives in content/art/*.png and can be drawn in any image editor.
# Rules: width and height are multiples of 8 (a tile), at most 4x4 tiles, and
# at most 4 colours per image (transparent counts as the background colour).
#
# "palette = auto" (the default) takes the four colours straight from the PNG:
# draw in whatever colours you like and the Game Boy shows those. Name a
# palette instead to force an existing one. Each distinct auto palette uses one
# of the console's 8 background palette slots.
#
# ASCII art still works: instead of "png = ...", write 16 rows of 16 characters
# using . - + # for the four colours. Handy for a quick tweak without an editor.

[sprite ball]
png = art/ball.png
# The console has eight background palettes and three go to the text, so the
# artwork can claim five. This little menu icon shares the built-in red
# "warn" palette instead of claiming a sixth; drop this line if you would
# rather it had its own colours and give one up elsewhere.
palette = warn
